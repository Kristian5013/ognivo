# Packages to hold for the length of an M0 baseline campaign (owner's decision 2026-10-30).
# unattended-upgrades updates the whole archive every morning (06:00-07:00). An NVIDIA userspace
# update without a reboot leaves the loaded kernel module mismatched (nvidia-smi fails: the next
# night ends as gpu-busy), and a podman/passt/runc update changes the container runtime between
# nights. The kernel meta-packages are held too, so no new kernel arrives without its matching
# NVIDIA module. Lines starting with # are comments; the rest are package names.
#
#   hold:    sudo apt-mark hold $(grep -v '^#' hold-packages.txt)
#   check:   apt-mark showhold
#   release: sudo apt-mark unhold $(grep -v '^#' hold-packages.txt)    (after the campaign)
#
# Versions on 2026-10-30: NVIDIA 580.178.04-0ubuntu1, kernel 7.3.0-9.9, podman 5.8.6+ds1-2,
